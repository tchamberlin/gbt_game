   135→  update(deltaTime: number): void {
   136→    if (!this.state.isStarted || this.state.isPaused) return;
   137→
   138→    // Handle game over explosion animation
   139→    if (this.state.isGameOver) {
   140→      this.explosionProgress += deltaTime * 0.5; // Explosion takes ~2 seconds
   141→      return;
   142→    }
   143→
   144→    // Update elapsed time and difficulty
   145→    this.state.elapsedTime += deltaTime;
   146→    this.state.difficultyLevel = 1 + Math.floor(this.state.elapsedTime / DIFFICULTY_INTERVAL) * 0.2;
   147→
   148→    // Update telescope (includes movement and jumping)
   149→    this.telescope.update(deltaTime);
   150→
   151→    // Track FRB count before update
   152→    const frbCountBefore = this.sources.getSources().filter((s) => s.type === 'frb').length;
   153→
   154→    // Update game objects
   155→    this.sources.update(deltaTime, this.state.difficultyLevel);
   156→    this.satellites.update(deltaTime, this.state.difficultyLevel);
   157→    this.groundhogs.update(deltaTime, this.state.difficultyLevel, this.telescope.state.x);
   158→
   159→    // Check for new FRBs
   160→    const frbCountAfter = this.sources.getSources().filter((s) => s.type === 'frb').length;
   161→    if (frbCountAfter > frbCountBefore) {
   162→      this.audio.playFRBAlert();
   163→    }
   164→
   165→    // Beam blanking is now controlled by mouse buttons (left=observe, right=radar)
   166→    // The state is managed in handleMouseDown/handleMouseUp
   167→
   168→    // Process beam collisions
   169→    const beam = this.telescope.getBeamState();
   170→    // Radar mode blocks observations like blanking does
   171→    const effectiveBlanked = this.state.isBeamBlanked || this.state.isRadarActive;
   172→    const collisions = processCollisions(
   173→      beam,
   174→      this.sources.getSources(),
   175→      this.satellites.getSatellites(),
   176→      deltaTime,
   177→      effectiveBlanked
   178→    );
   179→
   180→    // Handle observations - continuous scoring while observing
   181→    if (collisions.observedSources.length > 0) {
   182→      this.observationTickTimer += deltaTime;
   183→      if (this.observationTickTimer >= 0.2) {
   184→        this.observationTickTimer = 0;
   185→        this.audio.playObservationTick();
   186→      }
   187→
   188→      // Award points continuously while observing (small amount per frame)
   189→      for (const { source, deltaObservation } of collisions.observedSources) {
   190→        const continuousPoints = Math.floor(OBSERVATION_BONUS_RATE * deltaObservation);
   191→        if (continuousPoints > 0) {
   192→          this.state.score += continuousPoints;
   193→        }
   194→      }
   195→    } else {
   196→      this.observationTickTimer = 0;
   197→    }
   198→
   199→    // Handle completed sources - bonus points for completing observation
   200→    for (const source of collisions.completedSources) {
   201→      this.state.score += source.points;
   202→      this.audio.playSourceComplete(source.points);
   203→      this.addScorePopup(source.x, source.y, `+${source.points} ${source.type.toUpperCase()}`, '#00ff00');
   204→      this.sources.removeSource(source.id);
   205→    }
   206→
   207→    // Update radar disabled timer
   208→    if (this.state.radarDisabledTimer > 0) {
   209→      this.state.radarDisabledTimer -= deltaTime;
   210→      if (this.state.radarDisabledTimer < 0) {
   211→        this.state.radarDisabledTimer = 0;
   212→      }
   213→    }
   214→
   215→    // Handle radar mode - damage satellites and groundhogs in beam
   216→    // Radar is blocked when disabled timer is active
   217→    const radarBlocked = this.state.radarDisabledTimer > 0 || this.state.score <= 0;
   218→    if (this.state.isRadarActive && !radarBlocked) {
   219→      // Start radar sound
   220→      this.audio.startRadarSound();
   221→
   222→      // Calculate damage this frame
   223→      const damage = RADAR_DAMAGE_RATE * deltaTime;
   224→
   225→      // Damage satellites in beam
   226→      for (const satellite of collisions.satellitesInBeam) {
   227→        const destroyed = this.satellites.damageSatellite(satellite.id, damage);
   228→        if (destroyed) {
   229→          this.state.satellitesDestroyed++;
   230→          // Fine for destroying satellite
   231→          this.state.score = Math.max(0, this.state.score - SATELLITE_DESTRUCTION_FINE);
   232→          this.addScorePopup(
   233→            satellite.x,
   234→            satellite.y,
   235→            `-$${SATELLITE_DESTRUCTION_FINE} FINE!`,
   236→            '#ff4444'
   237→          );
   238→          // Spawn debris that falls to ground
   239→          this.satelliteDebris.push({
   240→            id: this.nextDebrisId++,
   241→            x: satellite.x,
   242→            y: satellite.y,
   243→            vy: 0,
   244→            isOnGround: false,
   245→          });
   246→          this.audio.playSatelliteDestroyed();
   247→        }
   248→      }
   249→
   250→      // Damage groundhogs in beam
   251→      for (const groundhog of this.groundhogs.getGroundhogs()) {
   252→        if (beamIntersectsGroundhog(beam, groundhog)) {
   253→          const destroyed = this.groundhogs.damageGroundhog(groundhog.id, damage);
   254→          if (destroyed) {
   255→            this.audio.playGroundhogDestroyed();
   256→          }
   257→        }
   258→      }
   259→
   260→      // Radar costs $10/second - accumulate fractional costs
   261→      this.radarCostAccumulator += RADAR_COST_RATE * deltaTime;
   262→      if (this.radarCostAccumulator >= 1) {
   263→        const cost = Math.floor(this.radarCostAccumulator);
   264→        this.radarCostAccumulator -= cost;
   265→        this.state.score = Math.max(0, this.state.score - cost);
   266→      }
   267→    } else {
   268→      // Stop radar sound when not active or blocked
   269→      this.audio.stopRadarSound();
   270→      this.radarCostAccumulator = 0;
   271→    }
   272→
   273→    // Handle satellites in beam - continuous penalty (only when not blanked and not radar)
   274→    if (collisions.satellitesInBeam.length > 0 && !this.state.isBeamBlanked && !this.state.isRadarActive) {
   275→      const penalty = Math.floor(SATELLITE_PENALTY_RATE * deltaTime * collisions.satellitesInBeam.length);
   276→      if (penalty > 0) {
   277→        this.state.score = Math.max(0, this.state.score - penalty);
   278→        this.satelliteFlash = 0.1;
   279→      }
   280→    }
   281→
   282→    // Reset ALL source timers when satellite enters beam (not blanked and not radar)
   283→    if (collisions.newSatellitesInBeam.length > 0 && !this.state.isBeamBlanked && !this.state.isRadarActive) {
   284→      // Reset ALL sources with any observation progress
   285→      for (const source of this.sources.getSources()) {
   286→        if (!source.isComplete && source.observedTime > 0) {
   287→          source.observedTime = 0;
   288→        }
   289→      }
   290→      this.addScorePopup(
   291→        this.renderer.width / 2,
   292→        100,
   293→        'ALL OBSERVATIONS RESET!',
   294→        '#ff8800'
   295→      );
   296→      this.audio.playSatelliteHit();
   297→    }
   298→
   299→    // Process groundhog collisions
   300→    const gbtBounds = this.telescope.getBaseBounds();
   301→    for (const groundhog of this.groundhogs.getGroundhogs()) {
   302→      if (groundhog.wasHit) continue;
   303→
   304→      const collisionResult = checkGroundhogCollision(
   305→        groundhog,
   306→        gbtBounds,
   307→        this.telescope.state.groundY,
   308→        this.telescope.state.y,
   309→        this.telescope.state.velocityY
   310→      );
   311→
   312→      if (collisionResult === 'stomp') {
   313→        // Jumping on groundhog kills it - bonus points!
   314→        groundhog.wasHit = true;
   315→        this.state.score += 50;
   316→        this.audio.playSourceComplete(50);
   317→        this.addScorePopup(groundhog.x, groundhog.y - 30, '+50 STOMP!', '#00ff00');
   318→      } else if (collisionResult === 'hit') {
   319→        // Groundhog hits wheels - damage!
   320→        groundhog.wasHit = true;
   321→        this.audio.playSatelliteHit();
   322→
   323→        // Get wheel position before damaging
   324→        const wheelPos = this.telescope.getWheelPosition(
   325→          this.telescope.state.wheels.findIndex(w => !w.damaged)
   326→        );
   327→
   328→        // Damage a wheel
   329→        const gameOver = this.telescope.damageWheel();
   330→
   331→        // Spawn wheel explosion at the damaged wheel's position
   332→        if (wheelPos) {
   333→          this.wheelExplosions.push({ x: wheelPos.x, y: wheelPos.y, progress: 0 });
   334→        }
   335→
   336→        this.addScorePopup(groundhog.x, groundhog.y - 30, 'WHEEL DESTROYED!', '#ff4444');
   337→
   338→        if (gameOver) {
   339→          this.state.isGameOver = true;
   340→          this.explosionProgress = 0;
   341→          this.audio.stopRadarSound();
   342→          this.saveHighScore();
   343→        }
   344→      }
   345→    }
   346→
   347→    // Update score popups
   348→    this.updateScorePopups(deltaTime);
   349→
   350→    // Update satellite flash
   351→    if (this.satelliteFlash > 0) {
   352→      this.satelliteFlash -= deltaTime;
   353→    }
   354→
   355→    // Update wheel explosions
   356→    for (const explosion of this.wheelExplosions) {
   357→      explosion.progress += deltaTime * 3; // Complete in ~0.33 seconds
   358→    }
   359→    this.wheelExplosions = this.wheelExplosions.filter(e => e.progress < 1);
   360→
   361→    // Update satellite debris (falling and collection)
   362→    const groundY = this.telescope.state.groundY;
   363→    const gbtX = this.telescope.state.x;
   364→    const gbtWidth = 80; // approximate width for collision
   365→    const debrisToRemove: number[] = [];
   366→
   367→    for (const debris of this.satelliteDebris) {
   368→      if (!debris.isOnGround) {
   369→        // Apply gravity
   370→        debris.vy += DEBRIS_GRAVITY * deltaTime;
   371→        debris.y += debris.vy * deltaTime;
   372→
   373→        // Check if hit ground
   374→        if (debris.y >= groundY - DEBRIS_SIZE / 2) {
   375→          debris.y = groundY - DEBRIS_SIZE / 2;
   376→          debris.vy = 0;
   377→          debris.isOnGround = true;
   378→        }
   379→      } else {
   380→        // Check collision with telescope for salvage
   381→        const dx = Math.abs(debris.x - gbtX);
   382→        if (dx < gbtWidth / 2 + DEBRIS_SIZE / 2) {
   383→          // Collected! Repair a wheel
   384→          const hasDamagedWheel = this.telescope.state.wheels.some(w => w.damaged);
   385→          if (hasDamagedWheel) {
   386→            this.telescope.repairWheel();
   387→            this.addScorePopup(debris.x, debris.y - 20, 'SALVAGE! +1 WHEEL', '#00ff00');
   388→            this.audio.playSourceComplete(100);
   389→          } else {
   390→            // All wheels intact, just give a small bonus
   391→            this.state.score += 100;
   392→            this.addScorePopup(debris.x, debris.y - 20, '+$100 SCRAP', '#88ff88');
   393→          }
   394→          debrisToRemove.push(debris.id);
   395→        }
   396→      }
   397→    }
   398→
   399→    // Remove collected debris
   400→    this.satelliteDebris = this.satelliteDebris.filter(d => !debrisToRemove.includes(d.id));
   401→
   402→    // Auto-purchase wheel repairs
   403→    const hasDamagedWheel = this.telescope.state.wheels.some(w => w.damaged);
   404→    if (hasDamagedWheel && this.state.score >= WHEEL_COST) {
   405→      if (this.telescope.repairWheel()) {
   406→        this.state.score -= WHEEL_COST;
   407→        this.addScorePopup(
   408→          this.renderer.width / 2,
   409→          150,
   410→          `-${formatDollars(WHEEL_COST)} WHEEL REPAIRED!`,
   411→          '#00ff88'
   412→        );
   413→        this.audio.playSourceComplete(100);
   414→      }
   415→    }
   416→
   417→    // Save high score periodically
   418→    this.saveHighScore();
   419→  }

   500→    this.heldMouseButtons.delete(button);

   502→    // Only respond to the button that was active

   504→
   505→    // Deactivate current action
   506→    if (button === 0) {
   507→      this.state.isBeamBlanked = true;
   508→    } else if (button === 2) {
   509→      this.state.isRadarActive = false;
   510→    }
   511→
   512→    // Check if other button is still held and transition to it
   513→    const otherButton = button === 0 ? 2 : 0;
   514→    if (this.heldMouseButtons.has(otherButton)) {
   515→      this.activeMouseButton = otherButton;
   516→      if (otherButton === 0) {
   517→        this.state.isBeamBlanked = false;
   518→      } else {
   519→        this.state.isRadarActive = true;
   520→      }
   521→    } else {
   522→      this.activeMouseButton = null;
   523→    }
